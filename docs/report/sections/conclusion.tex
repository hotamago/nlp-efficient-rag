\section{Kết luận}
\label{sec:conclusion}

\subsection{Trả lời câu hỏi nghiên cứu}

\begin{description}[leftmargin=2.6em,labelindent=0pt]
  \item[RQ1.] Ở mức nén $\approx$4$\times$, \textbf{chọn lọc theo câu} (Provence $\tau=0{,}1$, 4,6$\times$)
  và LongLLMLingua (4,3$\times$) là hai cách không làm giảm accuracy có ý nghĩa thống kê ở cả ba vị trí
  (64,0\% và 64,4\% so với 65,3\% của ngữ cảnh đầy đủ). Hai cách này ngang nhau về thống kê, nhưng
  Provence nén nhanh hơn 3 lần (200 so với 625 ms/câu trên GPU) và không cần LM sinh. Chọn lọc theo tài liệu (rerank top-5) rẻ hơn nhiều nhưng mất 2,5 điểm và giảm
  có ý nghĩa khi đoạn vàng ở cuối. Xoá theo token (LLMLingua-2 r=0,25) làm accuracy rơi về mức
  closed-book (38,3\% so với 38,8\%): phân tích lỗi cho thấy phần lớn thiệt hại đến từ văn bản bị xé
  vụn chứ không phải mất đáp án. Với reader hiện đại, đường cong vị trí vốn đã gần phẳng (chênh 5,8
  điểm) nên nén không còn tác dụng ``làm phẳng'' như báo cáo cho GPT-3.5.
  \item[RQ2.] Ở quy mô 50,7k đoạn, dense retriever nhỏ (bge-small, 33M tham số) vượt BM25 21 điểm
  hit@5; lượng tử int8 giảm chỉ mục 4$\times$ không mất chất lượng, nhị phân + rescore giảm 32$\times$
  chỉ mất 2,8 điểm. Ghép thành pipeline end-to-end (top-20 bge-small int8 $\to$ nén $\to$ reader 7B),
  accuracy đạt 66,0\%, ngang 20 tài liệu LITM; rerank top-5 hoặc Provence giảm 4$\times$ token chỉ
  mất 3,0--3,6 điểm (không có ý nghĩa).
  \item[RQ3.] Tái lập được hai kết luận của các công bố: BM25 kém xa dense, và LLMLingua-2 dùng đơn lẻ
  ở 4$\times$ sụt mạnh ($-$21 đến $-$30 điểm so với $-$11 đến $-$27 được công bố). Không tái lập được
  kết luận ``nén làm tăng accuracy khi đoạn vàng ở giữa'' của LongLLMLingua ($+17{,}1$ điểm với GPT-3.5,
  $+1{,}0$ không có ý nghĩa với Qwen2.5-7B): hiệu ứng đó phụ thuộc vào reader bị \emph{lost in the
  middle}, điều không còn đúng với LLM mở 7B thế hệ mới ở độ dài 3K token. Với reader nhỏ 1.5B vốn nhạy
  vị trí, hiệu ứng này xuất hiện trở lại: Provence tăng accuracy có ý nghĩa ở cả ba vị trí
  (Mục~\ref{sec:res-soft}).
  \item[RQ4.] Có. Bộ nén chạy trên CPU (rerank 0,2 s/câu, Provence 4,9 s/câu), reader gọi qua API;
  nén giảm 78\% token đầu vào và do đó giảm chi phí API tương ứng. GPU chỉ cần để rút ngắn độ trễ
  nén.
  \item[RQ5.] Chưa, ở quy mô của nhóm. Checkpoint công bố PISCO nén 18$\times$ nhưng mất 9--17 điểm so
  với văn bản trên cùng reader Mistral-7B (48,2\% so với 60,3\%); lọc trước bằng rerank top-5 nâng PISCO
  lên 58,5\% ở 72$\times$ nhưng vẫn kém văn bản rerank top-5 (62,9\%). Mô hình encode-once của nhóm
  (Qwen2.5-1.5B, 15k mẫu, 1 epoch) không vượt cổng go/no-go đặt trước: chỉ dùng vector nhớ thì ngang
  closed-book (25,7\% so với 26,2\% trên LITM), vì 8 vector không giữ được đáp án của từng đoạn (23\%
  so với 72\% khi đọc đoạn vàng dạng văn bản). Biến thể hybrid đạt 48,5\% ở 5,5$\times$, nhưng chỉ xấp xỉ
  chỉ đọc đoạn top-1 dạng văn bản (49,2\%). Ngược lại, trên reader 1.5B nhạy với vị trí, nén
  \emph{văn bản} bằng Provence tăng accuracy có ý nghĩa (+7,5 điểm trung bình). Với ngân sách nhỏ,
  chọn lọc văn bản vẫn là lựa chọn tốt hơn nén mềm.
\end{description}

\subsection{Hạn chế}

\begin{itemize}
  \item Chỉ một bộ dữ liệu (NQ-Open, câu trả lời ngắn) và một reader chính; kết luận về nén mức token
  có thể khác với tác vụ cần lập luận nhiều bước hoặc ngữ cảnh dài hơn nhiều (32K+).
  \item Chỉ số \emph{best subspan EM} phạt các câu trả lời đúng nhưng khác định dạng (ví dụ ngày
  tháng); một phần lỗi trong Mục~\ref{sec:res-errors} thuộc loại này.
  \item Các số liệu GPT-3.5 chỉ dùng để so sánh độ chênh, không so sánh tuyệt đối. Một phần nhỏ các lần
  chạy bổ sung (BM25/Dense top-5 ở vị trí 1/20, LongLLMLingua, rerank-5 của khối Mistral/PISCO ở vị trí 1/20) chạy trên
  máy thứ hai; kiểm tra chéo 60 câu cho câu trả lời trùng 56/60 và accuracy trùng 59/60, nhưng không đảm bảo trùng tuyệt đối.
  \item Mô hình nén mềm của nhóm được huấn luyện 1 epoch trên 15k mẫu với một GPU 12GB, ít hơn nhiều
  so với dữ liệu và tài nguyên của PISCO; kết luận thất bại chỉ đúng ở quy mô này (một epoch, 8
  vector/đoạn, LLM 1.5B).
\end{itemize}

\subsection{Hướng phát triển}

\begin{itemize}
  \item Kết hợp hai nhánh hiệu quả nhất: chọn lọc theo câu (Provence) cho tài liệu được xếp hạng cao
  và bộ nhớ nén cho phần còn lại -- tương tự cơ chế trộn token chọn lọc và token nén trong attention
  thưa của DeepSeek.
  \item Đánh giá trên ngữ cảnh dài hơn (50--100 tài liệu) và các bộ dữ liệu đa bước (HotpotQA,
  MuSiQue), nơi reader còn bị ảnh hưởng bởi vị trí.
  \item Chưng cất Provence sang mô hình nhỏ hơn để giảm độ trễ trên CPU xuống dưới 1 s/câu.
\end{itemize}
